\begin{frame}[t]{State space and algebra}
\hypertarget{main:states}{}
\hypertarget{main:algebras}{}
\fontsize{9}{11}\selectfont
\begin{columns}[T,onlytextwidth]
  \begin{column}{.50\textwidth}
    \heading{Vector space}
    \begin{equation*}
      V=\mathbb R^d,\qquad x=(x_1,\ldots,x_d).
    \end{equation*}
    Addition and scalar multiplication:
    \begin{equation*}
      \begin{aligned}
        (x+y)_i&=x_i+y_i,\\
        (\lambda x)_i&=\lambda x_i.
      \end{aligned}
    \end{equation*}
    \heading{Endomorphism algebra}
    \begin{equation*}
      \begin{aligned}
        \operatorname{End}(V)&=\{A:V\to V:\ A\text{ linear}\}\\
          &\cong M_d(\mathbb R).
      \end{aligned}
    \end{equation*}
    \begin{equation*}
      AB=A\circ B,\qquad I_Vx=x.
    \end{equation*}
    Associative composition; unit $I_V$.
  \end{column}
  \begin{column}{.45\textwidth}
    \heading{Euclidean geometry}
    \flowdiagram{state}
    \begin{equation*}
      \|x\|=\sqrt{x_1^2+\cdots+x_d^2}.
    \end{equation*}
    The norm defines distance, continuity and differentiability.
  \end{column}
\end{columns}
\slidecite{Manuscript \S\,1.1 and \S\,1.2.1. \hyperlink{app:algebras}{Further algebra operations: Appendix A1.}}

\begin{frame}[t]{From a velocity field to an ODE}
\hypertarget{main:ode}{}
\fontsize{9}{11}\selectfont
A fluid element is carried by the local fluid velocity. Its position $x_t\in V$
evolves from an initial position $x$ at time $s$ according to
\begin{equation*}
  \begin{cases}
    \dot x_t=v_t(x_t),&t\in I,\\
    x_s=x.
  \end{cases}
\end{equation*}
Here $I\subseteq\mathbb R$ is an open time interval, with $(s,x)\in I\times V$.
\heading{Velocity field}
\begin{equation*}
  \begin{aligned}
    v &: I\times V\longrightarrow V,\qquad (t,z)\longmapsto v_t(z),\\
    v_t &: V\longrightarrow V,\qquad z\longmapsto v_t(z).
  \end{aligned}
\end{equation*}
At a fixed time $t$, $v_t$ assigns a velocity to every position $z$.
\heading{Trajectory}
\begin{equation*}
  x_{\cdot}:I\longrightarrow V,\qquad t\longmapsto x_t,
  \qquad x_{\cdot}\in C^1(I,V).
\end{equation*}
The solution traces one fluid element through time. Its tangent $\dot x_t$
equals the field evaluated at the current position, $v_t(x_t)$.
\slidecite{Manuscript \S\,1.2.1, fluid-motion background and ODE initial-value problems.}
\end{frame}

\section{Project 9.1: affine dynamics}

\begin{frame}[t]{Project 9.1: the initial-value problem}
\hypertarget{main:ivp}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.54\textwidth}
    \heading{Given data}
    An open interval $I\subseteq\mathbb R$, with
    \begin{equation*}
      \begin{aligned}
        A&\in C(I,\operatorname{End}(V)),\\
        B&\in C(I,V).
      \end{aligned}
    \end{equation*}
    For $(s,x)\in I\times V$, solve
    \begin{equation*}
      \begin{cases}
        \dot x_t=A_tx_t+B_t,&t\in I,\\
        x_s=x.
      \end{cases}
    \end{equation*}
    Continuous coefficients give a unique trajectory on all of $I$.
  \end{column}
  \begin{column}{.41\textwidth}
    \heading{Trajectory formulation}
    \flowdiagram{trajectory}
    \begin{equation*}
      \mathsf D_A\gamma=B,\qquad E_s\gamma=x.
    \end{equation*}
    The unknown is a complete trajectory.
  \end{column}
\end{columns}
\begin{equation*}
  \mathcal X=C^1(I,V),\qquad \mathcal Y=C(I,V),\qquad
  (\mathsf M_A\gamma)_t=A_t\gamma_t.
\end{equation*}
\slidecite{Assessment 9.1; Manuscript Example 1.10. \hyperlink{app:existence}{Existence and uniqueness: Appendix A2.}}
\end{frame}

\begin{frame}[t]{Homogeneous evolution by evaluation}
\hypertarget{main:homogeneous}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.54\textwidth}
    \heading{The homogeneous kernel}
    \begin{equation*}
      \mathcal H=\ker\mathsf D_A.
    \end{equation*}
    Evaluation identifies a trajectory with its state:
    \begin{equation*}
      E_s:\mathcal H\to V,\qquad E_s(\eta)=\eta_s.
    \end{equation*}
    Existence and uniqueness make $E_s$ a linear isomorphism.
    \begin{equation*}
      S_s=E_s^{-1},\qquad U(t,s)=E_tS_s.
    \end{equation*}
  \end{column}
  \begin{column}{.41\textwidth}
    \heading{Recover, then evaluate}
    \flowdiagram{linear-evolution}
    \medskip
    \begin{equation*}
      \begin{aligned}
        \partial_t U(t,s)&=A_tU(t,s),\\
        U(s,s)&=I_V.
      \end{aligned}
    \end{equation*}
  \end{column}
\end{columns}
\slidecite{Manuscript Example 1.10. \hyperlink{app:evaluation}{Evaluation isomorphism and composition: Appendix A3.}}
\end{frame}

\begin{frame}[t]{The affine solution and its flow}
\hypertarget{main:affine}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.55\textwidth}
    \heading{Homogeneous and forced response}
    \begin{equation*}

        (af+bg)(x)&=af(x)+bg(x),\\
        (fg)(x)&=f(x)g(x),\\

\begin{frame}[t]{Appendix A2: existence and uniqueness}
\hypertarget{app:existence}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.53\textwidth}
    \heading{Bounds on a compact time window}
    For $v_t(y)=A_ty+B_t$ and compact $J\subset I$, let
    \begin{equation*}
      L_J=\sup_{t\in J}\|A_t\|,\qquad
      C_J=\sup_{t\in J}\|B_t\|.
    \end{equation*}
    Both constants are finite, and
    \begin{equation*}
      \begin{aligned}
        \|v_t(y)-v_t(z)\|&\leq L_J\|y-z\|,\\
        \|v_t(y)\|&\leq C_J+L_J\|y\|.
      \end{aligned}
    \end{equation*}
  \end{column}
  \begin{column}{.42\textwidth}
    \heading{Continuation argument}
    \begin{enumerate}\setlength{\itemsep}{7pt}
      \item Local Lipschitz continuity gives a unique local solution.
      \item Linear growth prevents finite-time escape on $J$.
      \item Uniqueness joins the solutions on overlapping windows.
    \end{enumerate}
    The trajectory is therefore uniquely defined on all of $I$.
  \end{column}
\end{columns}
\slidecite{Manuscript Example 1.10 and Appendix A.1. \hyperlink{main:ivp}{Return to the initial-value problem.}}
\end{frame}

\begin{frame}[t]{Appendix A3: evaluation and evolution}
\hypertarget{app:evaluation}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.52\textwidth}
    \heading{Homogeneous trajectories}
    \begin{equation*}
      \mathcal H=\{\eta\in C^1(I,V):\dot\eta_t=A_t\eta_t\}.
    \end{equation*}
    Evaluation $E_s(\eta)=\eta_s$ is linear.
    \begin{itemize}\setlength{\itemsep}{4pt}
      \item Existence makes it onto.
      \item Uniqueness makes it one-to-one.
    \end{itemize}
    \begin{equation*}
      S_s=E_s^{-1},\qquad \mathcal H\cong V,
      \qquad\dim\mathcal H=d.
    \end{equation*}
  \end{column}
  \begin{column}{.43\textwidth}
    \heading{Inverse evaluation}
    \flowdiagram{evaluation}
    \begin{equation*}
      E_sS_s=I_V,\qquad S_sE_s=I_{\mathcal H}.
    \end{equation*}
  \end{column}
\end{columns}
\heading{Composition of the homogeneous evolution}
\begin{equation*}
  U(t,r)U(r,s)=E_tS_rE_rS_s=E_tS_s=U(t,s).
\end{equation*}
\slidecite{Manuscript Example 1.10. \hyperlink{main:homogeneous}{Return to homogeneous evolution.}}
\end{frame}

\begin{frame}[t]{Appendix A4: the affine solution family}
\hypertarget{app:affine}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.52\textwidth}
    \heading{A particular solution shifts the kernel}
    Suppose $\mathsf D_A\gamma_{\mathrm p}=B$. Then
    \begin{equation*}
      \begin{aligned}
        \mathsf D_A\gamma=B
          &\iff\mathsf D_A(\gamma-\gamma_{\mathrm p})=0\\
          &\iff\gamma-\gamma_{\mathrm p}\in\mathcal H.
      \end{aligned}
    \end{equation*}
    Hence
    \begin{equation*}
      \mathcal S_B=\gamma_{\mathrm p}+\mathcal H.
    \end{equation*}
  \end{column}
  \begin{column}{.43\textwidth}
    \heading{The initial condition selects one path}
    Write $\gamma=\gamma_{\mathrm p}+\eta$. Then
    \begin{equation*}
      E_s\eta=x-\gamma_{\mathrm p}(s).
    \end{equation*}
    Since $E_s$ is an isomorphism,
    \begin{equation*}
      \gamma=\gamma_{\mathrm p}
        +S_s\bigl(x-\gamma_{\mathrm p}(s)\bigr).
    \end{equation*}
    Taking $\gamma_{\mathrm p}(s)=0$ gives
    \begin{equation*}
      \gamma=\gamma_{\mathrm p}+S_sx.
    \end{equation*}
  \end{column}
\end{columns}
\slidecite{Manuscript Example 1.10. \hyperlink{main:affine}{Return to the affine solution.}}
\end{frame}

\begin{frame}[t]{Appendix A5: variation of constants}
\hypertarget{app:variation}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.48\textwidth}
    \heading{1. Vary the homogeneous coefficient}
    \begin{equation*}
      \gamma_{\mathrm p}(t)=U(t,s)c(t),\qquad c(s)=0.
    \end{equation*}
    \heading{2. Apply the trajectory operator}
    \begin{equation*}
      \begin{aligned}
        \dot\gamma_{\mathrm p}-A_t\gamma_{\mathrm p}
          &=U(t,s)\dot c(t),\\
        \dot c(t)&=U(s,t)B_t.
      \end{aligned}
    \end{equation*}
  \end{column}
  \begin{column}{.47\textwidth}
    \heading{3. Integrate and compose}
    \begin{equation*}
      c(t)=\int_s^t U(s,r)B_r\,\mathrm dr.
    \end{equation*}
    Use $U(t,s)U(s,r)=U(t,r)$ to obtain
    \begin{equation*}
      \begin{aligned}
        \gamma_{\mathrm p}(t)
          &=U(t,s)\int_s^t U(s,r)B_r\,\mathrm dr\\
          &=\int_s^t U(t,r)B_r\,\mathrm dr.
      \end{aligned}
    \end{equation*}
  \end{column}
\end{columns}
\medskip
\begin{equation*}
  x_t=U(t,s)x+\int_s^t U(t,r)B_r\,\mathrm dr.
\end{equation*}
\slidecite{Manuscript Example 1.10. \hyperlink{main:affine}{Return to the solution formula.}}
\end{frame}

\begin{frame}[t]{Appendix A6: verification of the solution}
\hypertarget{app:verification}{}
\heading{Initial condition}
\begin{equation*}
  x_s=U(s,s)x+\int_s^s U(s,r)B_r\,\mathrm dr=x.
\end{equation*}
\heading{Differentiation}
Since $\partial_tU(t,r)=A_tU(t,r)$, Leibniz's rule gives
\begin{equation*}
  \begin{aligned}
    \dot x_t
      &=A_tU(t,s)x+U(t,t)B_t+\int_s^t A_tU(t,r)B_r\,\mathrm dr\\
      &=A_t\left(U(t,s)x+\int_s^t U(t,r)B_r\,\mathrm dr\right)+B_t\\
      &=A_tx_t+B_t.
  \end{aligned}
\end{equation*}
\heading{Uniqueness}
The difference of two solutions lies in $\mathcal H$ and has zero value at $s$.
Injectivity of $E_s$ makes that difference the zero trajectory.
\slidecite{Manuscript Example 1.10. \hyperlink{main:affine}{Return to the solution and its flow map.}}
\end{frame}

\begin{frame}[t]{Appendix A7: general evaluation and composition}
\hypertarget{app:general}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.49\textwidth}
    \heading{Evaluation is a bijection}
    \begin{equation*}
      \mathcal S_v=\{\gamma\in C^1(I,V):\dot\gamma_t=v_t(\gamma_t)\}.
    \end{equation*}
    On the solution set, $E_s(\gamma)=\gamma_s$.
    Existence gives surjectivity; uniqueness gives injectivity.
    \begin{equation*}
      \Phi_{s,t}=E_t\circ E_s^{-1}.
    \end{equation*}
  \end{column}
  \begin{column}{.46\textwidth}
    \heading{Recover, then evaluate}
    \flowdiagram{general-evaluation}
  \end{column}
\end{columns}
\heading{Cancellation at an intermediate time}
\begin{equation*}
  \begin{aligned}
    \Phi_{r,t}\circ\Phi_{s,r}
      &=(E_t\circ E_r^{-1})\circ(E_r\circ E_s^{-1})\\
      &=E_t\circ E_s^{-1}=\Phi_{s,t}.
  \end{aligned}
\end{equation*}
\begin{equation*}
  \Phi_{s,s}=I_V,\qquad \Phi_{s,t}^{-1}=\Phi_{t,s}.
\end{equation*}
\slidecite{Manuscript Appendix A.2. \hyperlink{main:general}{Return to general flows}; \hyperlink{main:laws}{return to the flow laws}.}
\end{frame}

\begin{frame}[t]{Appendix A8: autonomy and completeness}
\hypertarget{app:autonomous}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.48\textwidth}
    \heading{Autonomous field}
    \begin{equation*}
      v_t=v,\qquad \Phi_{s,t}=\phi_{t-s}.
    \end{equation*}
    Uniqueness gives time-translation invariance.
    If all trajectories exist for every real time,
    \begin{equation*}
      \begin{aligned}
        \phi_{\tau+\sigma}&=\phi_\tau\circ\phi_\sigma,\\
        \phi_0&=I_V,\\
        \phi_{-\tau}&=\phi_\tau^{-1}.
      \end{aligned}
    \end{equation*}
    The maps form a one-parameter group.
  \end{column}
  \begin{column}{.47\textwidth}
    \heading{Time-dependent field}
    \begin{equation*}
      v_t=v(t,\cdot).
    \end{equation*}
    Both initial and terminal times matter. The valid composition law is
    \begin{equation*}
      \Phi_{r,t}\circ\Phi_{s,r}=\Phi_{s,t}.
    \end{equation*}
    The family $\Phi_{0,t}$ generally does not have the same one-parameter group law.
  \end{column}
\end{columns}
\slidecite{Manuscript Theorem 1.12 and Appendix A.2. \hyperlink{main:laws}{Return to the main flow laws.}}
\end{frame}

\begin{frame}[t]{Appendix A9: time reversal}
\hypertarget{app:reversal}{}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.54\textwidth}
    \heading{Reverse a finite time window}
    For a trajectory on $[0,T]$, set
    \begin{equation*}
      y_t=x_{T-t},\qquad0\leq t\leq T.
    \end{equation*}
    The chain rule gives
    \begin{equation*}
      \begin{aligned}
        \dot y_t&=-\dot x_{T-t}\\
          &=-v_{T-t}(x_{T-t})\\
          &=-v_{T-t}(y_t).
      \end{aligned}
    \end{equation*}
    Thus
    \begin{equation*}
      \widetilde v_t=-v_{T-t},\qquad y_0=x_T.
    \end{equation*}
  \end{column}
