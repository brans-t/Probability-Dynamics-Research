% Shared 4:3 serif Beamer style with the project's blue Madrid palette.
\usetheme{Madrid}
\usefonttheme{serif}

\usepackage[english]{babel}
\usepackage{amsmath,amssymb,mathtools}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,calc}

\definecolor{accent}{RGB}{55,55,180}
\definecolor{diagramred}{RGB}{178,65,56}

\setbeamercolor{structure}{fg=accent}
\setbeamercolor{frametitle}{fg=white,bg=accent}
\setbeamercolor{title}{fg=white,bg=accent}
\setbeamercolor{palette primary}{fg=white,bg=accent}
\setbeamercolor{palette secondary}{fg=white,bg=accent!88!black}
\setbeamercolor{palette tertiary}{fg=white,bg=accent!74!black}

\setbeamertemplate{navigation symbols}{}

% Let Beamer manage the page furniture and body flow. Diagrams alone use TikZ.
\setbeamertemplate{headline}{}
\setbeamertemplate{frametitle}[default][left]
\setbeamerfont{frametitle}{size=\fontsize{13}{15.5}\selectfont}
\setbeamerfont{title}{size=\fontsize{20}{24}\selectfont}
\setbeamerfont{subtitle}{size=\fontsize{11}{14}\selectfont}
\setbeamertemplate{section in toc}[sections numbered]
\setbeamertemplate{title page}{%
  \vbox{}\vfill
  \begin{centering}
    {\usebeamerfont{title}\usebeamercolor[fg]{structure}\inserttitle\par}
    \vspace{12pt}
    {\usebeamerfont{subtitle}\insertsubtitle\par}
    \vspace{22pt}
    {\insertauthor\par}
    \vspace{7pt}
    {\small\insertdate\par}
  \end{centering}
  \vfill
}
\setbeamerfont{footline}{size=\fontsize{6}{7}\selectfont}
\setbeamercolor{footline}{fg=white,bg=accent!85!black}
\setbeamerfont{framesource}{size=\fontsize{5.8}{7}\selectfont}
\setbeamercolor{framesource}{fg=gray,bg=white}
\newcommand{\currentframesource}{}
% Beamer records both totals at \appendix; keep the underlying frame counter
% continuous so that navigation and animation destinations remain unique.
\makeatletter
\newcommand{\slidepagenumber}{%
  \ifbeamer@inappendix
    A-\insertframenumberinappendix\,/\,\insertappendixframenumber
  \else
    \insertframenumber\,/\,\insertmainframenumber
  \fi}
\makeatother
\setbeamertemplate{footline}{%
  \begin{beamercolorbox}[wd=\paperwidth,ht=14pt,dp=4pt,leftskip=18pt,rightskip=18pt]{framesource}%
    \usebeamerfont{framesource}\strut\currentframesource\strut
  \end{beamercolorbox}%
  \leavevmode\hbox{%
    \begin{beamercolorbox}[wd=.2\paperwidth,ht=8pt,dp=3.5pt,leftskip=12pt]{footline}%
      \usebeamerfont{footline}\strut H. Zhang
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.6\paperwidth,ht=8pt,dp=3.5pt,center]{footline}%
      \usebeamerfont{footline}\strut\insertsectionhead
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.2\paperwidth,ht=8pt,dp=3.5pt,right,rightskip=12pt]{footline}%
      \usebeamerfont{footline}\strut\slidepagenumber
    \end{beamercolorbox}%
  }\vskip0pt%
}
